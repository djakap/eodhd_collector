# Prefect (self-hosted server + worker)
# Pinned to match the server image tag in docker-compose.yml (both 3.6.18).
# Keep the two in lockstep: server must be >= this client version, or Prefect
# warns of "unexpected behavior". Do NOT bump to 3.7.x on the current SQLite
# backend — it causes persistent "database is locked" contention (see the
# server image comment in docker-compose.yml). Bump both together after moving
# to Postgres, then rebuild the worker.
prefect==3.6.18
# Prefect 3.6.18 imports the importlib_metadata backport unconditionally in
# workers/base.py, but its dependency marker skips it on Python 3.12+, so a
# plain `pip install prefect==3.6.18` omits it and the worker crash-loops with
# ModuleNotFoundError. The official prefecthq server image bundles it; our
# worker build must add it explicitly. Drop this when moving off 3.6.18.
importlib_metadata>=6.0
